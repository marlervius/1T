\seksjon{2.2}{Likningssystemer}
\undertittel{To ukjente, to likninger. Regn det ut og se det på grafen.}
\maal{Løse likningssystemer med to ukjente ved regning og grafisk, og avgjøre hvor mange løsninger de har}
\kompetansemaal{4}

\hvakanjeg{
\kan{1}{løse et likningssystem med innsettingsmetoden}
\kan{1}{løse et likningssystem med addisjonsmetoden}
\kan{2}{velge metode og begrunne valget}
\kan{2}{kontrollere svaret i begge likningene}
\kanskille
\kan{1}{forklare hvorfor løsningen er skjæringspunktet mellom to linjer}
\kan{2}{avgjøre om systemet har én, ingen eller uendelig mange løsninger}
\kan{2}{tolke $0 = 0$ og en usann påstand når jeg legger sammen likninger}
}
