\seksjon{2.4}{Ulikheter}
\undertittel{Løs ulikheter, snu tegnet riktig og les svaret av en fortegnslinje.}
\maal{Løse lineære ulikheter og andregradsulikheter og knytte dem til grafen}
\kompetansemaal{4,6}

\hvakanjeg{
\kan{1}{løse lineære ulikheter og snu tegnet ved negativt tall}
\kan{1}{skrive svaret som ulikhet, på tallinje og som intervall}
\kan{1}{løse en dobbel ulikhet}
\kanskille
\kan{2}{løse andregradsulikheter med fortegnslinje}
\kan{2}{forklare sammenhengen mellom ulikheten og grafen til en parabel}
\kan{2}{avgjøre når en andregradsulikhet gjelder for alle $x$ eller aldri}
}
