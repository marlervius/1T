\seksjon{2.1}{Likninger og formler}
\undertittel{Uttrykk, likning, identitet og funksjon. Løs, sett opp og snu.}
\maal{Skille mellom uttrykk, likning, identitet og funksjon, løse likninger med brøk og sette opp og snu formler}
\kompetansemaal{5,3,4}

\hvakanjeg{
\kan{1}{forklare forskjellen på uttrykk, likning, identitet og funksjon}
\kan{1}{vise at noe ikke er en identitet med ett moteksempel}
\kan{1}{tolke $0 = 0$ og $0 = 7$ når jeg løser en likning}
\kanskille
\kan{2}{løse likninger med brøk og kontrollere svaret}
\kan{2}{sette opp en formel ut fra en situasjon}
\kan{2}{løse en formel med hensyn på en annen variabel}
}
