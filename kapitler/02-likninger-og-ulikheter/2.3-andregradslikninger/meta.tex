\seksjon{2.3}{Andregradslikninger}
\undertittel{Faktoriser eller bruk abc-formelen. Velg metoden etter at du har sett på likningen.}
\maal{Løse andregradslikninger ved faktorisering og med abc-formelen, og avgjøre hvor mange løsninger de har}
\kompetansemaal{6,4}

\hvakanjeg{
\kan{1}{løse en likning med nullproduktregelen}
\kan{1}{velge mellom felles faktor, kvadratrot, kvadratsetning og sum--produkt}
\kan{2}{forklare hvorfor vi ikke deler på $x$ når $x$ kan være 0}
\kan{2}{forklare hvorfor nullproduktregelen bare gjelder for 0}
\kanskille
\kan{1}{løse en andregradslikning med abc-formelen}
\kan{2}{bruke diskriminanten til å finne antall løsninger}
\kan{2}{forklare sammenhengen mellom løsninger, nullpunkter og faktorisering}
}
