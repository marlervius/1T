\seksjon{3.3}{Polynomfunksjoner}
\undertittel{Polynomdivisjon, faktorisering og grafer.}
\maal{Forklare og utføre polynomdivisjon, faktorisere polynomer og løse likninger og ulikheter av grad tre}
\kompetansemaal{11,9}

\hvakanjeg{
\kan{1}{utføre polynomdivisjon og kontrollere svaret}
\kan{1}{bruke faktorsetningen til å avgjøre om $(x - a)$ er en faktor}
\kan{2}{faktorisere et tredjegradspolynom fullstendig}
\kan{2}{forklare hvorfor resten ved divisjon med $(x - a)$ er $P(a)$}
\kanskille
\kan{1}{finne grad, ledende ledd og nullpunkter til en polynomfunksjon}
\kan{2}{skissere grafen ut fra nullpunkter og ledende ledd}
\kan{2}{løse likninger og ulikheter av grad tre med faktorisering og fortegnslinje}
\kan{2}{forklare hva et dobbelt nullpunkt betyr for grafen}
}
