\seksjon{3.4}{Rasjonale funksjoner}
\undertittel{Finn asymptotene. Løs likninger og ulikheter med brøk.}
\maal{Finne asymptoter og hull i grafen til rasjonale funksjoner, og løse rasjonale likninger og ulikheter}
\kompetansemaal{9,11,4}

\hvakanjeg{
\kan{1}{finne vertikal og horisontal asymptote til en brøk med $x$ i teller og nevner}
\kan{1}{bruke polynomdivisjon til å forklare asymptotene}
\kan{1}{se når en graf har et hull i stedet for en asymptote}
\kanskille
\kan{2}{løse likninger med brøk og kontrollere svaret mot vilkåret}
\kan{2}{løse ulikheter med brøk ved hjelp av fortegnslinje}
\kan{2}{forklare hvorfor vi ikke ganger ulikheter med et uttrykk med $x$}
}
