\seksjon{3.6}{Modellering}
\undertittel{Fra tall til modell. Og hvor langt kan vi stole på den?}
\maal{Lage modeller med regresjon, tolke dem og vurdere om de gir mening}
\kompetansemaal{7,8,3}

\hvakanjeg{
\kan{1}{velge modelltype ut fra en tabell og situasjonen}
\kan{1}{bruke regresjon i et digitalt verktøy}
\kan{2}{tolke konstantene i en modell med enhet}
\kan{2}{forklare hvorfor $x$ bør være «år etter» og ikke årstallet}
\kanskille
\kan{1}{si hvor en modell gir mening (gyldighetsområde)}
\kan{2}{hente ut tall fra en kort tekst og vurdere en påstand}
\kan{2}{forklare hvorfor høy forklaringsgrad ikke er nok}
\kan{2}{skrive «ifølge modellen» og nevne forbehold}
}
