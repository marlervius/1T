\seksjon{3.5}{Eksponential- og potensfunksjoner}
\undertittel{Faste faktorer gir vekst. Se forskjellen på $2^x$ og $x^2$.}
\maal{Bruke og tolke eksponentialfunksjoner og potensfunksjoner, og skille dem fra lineære funksjoner}
\kompetansemaal{9,7}

\hvakanjeg{
\kan{1}{finne vekstfaktoren når veksten er oppgitt i prosent}
\kan{1}{tolke $a$ og $b$ i $f(x) = a \cdot b^x$}
\kan{2}{finne en eksponentialfunksjon gjennom to punkter}
\kan{2}{forklare forskjellen på lineær og eksponentiell vekst}
\kanskille
\kan{1}{se forskjell på $2^x$ og $x^2$}
\kan{2}{si hva som skjer med $f(x) = a \cdot x^b$ når $x$ dobles}
\kan{2}{avgjøre om en tabell er lineær, eksponentiell eller en potensfunksjon}
\kan{2}{forklare hvorfor $x^{\tfrac12}$, $x$ og $x^2$ bytter rekkefølge ved $x = 1$}
}
