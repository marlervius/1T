\seksjon{1.1}{Potenser}
\undertittel{Bruk reglene. Forstå hvorfor de virker.}
\maal{Regne med potensreglene, null og negative eksponenter og standardform}
\kompetansemaal{}

\hvakanjeg{
\kan{1}{skrive produkter og brøker av potenser som én potens}
\kan{1}{forklare hvorfor $a^m \cdot a^n = a^{m+n}$}
\kan{1}{se forskjell på $(-2)^4$ og $-2^4$}
\kanskille
\kan{2}{forklare hvorfor $a^0 = 1$ og $a^{-n} = \frac{1}{a^n}$}
\kan{2}{forenkle uttrykk med negative eksponenter}
\kan{2}{skrive og regne med tall i standardform}
}
