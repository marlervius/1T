\seksjon{1.4}{Rasjonale uttrykk}
\undertittel{Brøker med bokstaver. Faktoriser først, forkort etterpå.}
\maal{Forkorte, gange, dele, addere og subtrahere rasjonale uttrykk og skrive vilkår}
\kompetansemaal{4,11}

\hvakanjeg{
\kan{1}{forklare hva et rasjonalt uttrykk er, og når det ikke er definert}
\kan{1}{faktorisere teller og nevner og forkorte brøken}
\kan{1}{skrive vilkår for at nevneren ikke er null}
\kan{2}{gange og dele rasjonale uttrykk}
\kanskille
\kan{2}{finne minste fellesnevner ved å faktorisere nevnerne}
\kan{2}{addere og subtrahere rasjonale uttrykk}
\kan{2}{regne riktig med minus foran en brøk}
}
