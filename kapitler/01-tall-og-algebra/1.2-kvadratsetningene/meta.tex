\seksjon{1.2}{Kvadratsetningene og faktorisering}
\undertittel{Gang ut, faktoriser og bruk mønstrene baklengs.}
\maal{Gange ut med kvadratsetningene, faktorisere og kontrollere}
\kompetansemaal{6}

\hvakanjeg{
\kan{1}{gange ut $(a+b)^2$, $(a-b)^2$ og $(a+b)(a-b)$}
\kan{1}{forklare med en figur hvorfor $(a+b)^2 = a^2 + 2ab + b^2$}
\kan{1}{bruke kvadratsetningene til hoderegning}
\kanskille
\kan{2}{trekke ut felles faktor}
\kan{2}{faktorisere med kvadratsetningene baklengs}
\kan{2}{kontrollere en faktorisering ved å gange ut}
}
