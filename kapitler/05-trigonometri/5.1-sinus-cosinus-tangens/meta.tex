\seksjon{5.1}{Sinus, cosinus og tangens}
\undertittel{Forholdet mellom sidene i en rettvinklet trekant, og videre til stumpe vinkler.}
\maal{Bruke sinus, cosinus og tangens til å finne sider og vinkler, og regne med eksakte verdier og enhetssirkelen}
\kompetansemaal{12}

\hvakanjeg{
\kan{1}{skrive definisjonene av sinus, cosinus og tangens ut fra sidene i en rettvinklet trekant}
\kan{1}{forklare hvorfor forholdene bare avhenger av vinkelen}
\kan{1}{finne sider og vinkler med kalkulator}
\kanskille
\kan{2}{bruke eksakte verdier for $30^\circ$, $45^\circ$ og $60^\circ$}
\kan{2}{forklare enhetssirkelen og finne sinus og cosinus til stumpe vinkler}
\kan{2}{finne begge vinklene mellom $0^\circ$ og $180^\circ$ når sinus er gitt}
}
