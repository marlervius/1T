\seksjon{5.2}{Arealsetningen}
\undertittel{Areal av en vilkårlig trekant ut fra to sider og vinkelen mellom dem.}
\maal{Begrunne og bruke arealsetningen til å finne areal, sider og vinkler i trekanter}
\kompetansemaal{13,12}

\hvakanjeg{
\kan{1}{finne arealet av en trekant når to sider og vinkelen mellom dem er gitt}
\kan{1}{begrunne arealsetningen for både spisse og stumpe vinkler}
\kan{1}{bruke arealsetningen også når vinkelen er $90^\circ$ eller større}
\kanskille
\kan{2}{finne en side når arealet og de andre målene er kjent}
\kan{2}{finne begge mulige vinkler når arealet er kjent}
\kan{2}{finne arealet av en firkant ved å dele den i trekanter}
}
