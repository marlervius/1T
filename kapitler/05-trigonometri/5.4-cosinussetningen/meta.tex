\seksjon{5.4}{Cosinussetningen}
\undertittel{Finn sider og vinkler når sinussetningen ikke holder. Velg riktig setning.}
\maal{Begrunne og bruke cosinussetningen, og velge riktig setning for en trekant}
\kompetansemaal{13,12}

\hvakanjeg{
\kan{1}{finne den tredje siden når to sider og vinkelen mellom dem er gitt}
\kan{1}{finne en vinkel når alle tre sidene er gitt}
\kan{1}{begrunne cosinussetningen for både spisse og stumpe vinkler}
\kanskille
\kan{2}{velge mellom sinussetningen og cosinussetningen ut fra det som er kjent}
\kan{2}{løse SSV-tilfellet med cosinussetningen og en andregradslikning}
\kan{2}{kontrollere svaret med vinkelsum og største side mot største vinkel}
}
