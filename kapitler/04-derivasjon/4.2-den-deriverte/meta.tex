\seksjon{4.2}{Den deriverte}
\undertittel{Fra tabell med mindre og mindre $h$ til en eksakt verdi.}
\maal{Gjøre rede for den deriverte og utlede de første derivasjonsreglene fra definisjonen}
\kompetansemaal{10,2}

\hvakanjeg{
\kan{1}{forklare hva en grenseverdi er, og hvorfor vi ikke kan sette $h = 0$ med en gang}
\kan{1}{regne ut $f'(a)$ og $f'(x)$ fra definisjonen}
\kan{1}{forklare forskjellen på $f(a)$ og $f'(a)$}
\kanskille
\kan{2}{utlede at $(x^2)' = 2x$ og $(x^3)' = 3x^2$ fra definisjonen}
\kan{2}{forklare hvorfor $(f + g)' = f' + g'$}
\kan{2}{bruke mønsteret $(x^n)' = nx^{n-1}$ og si hva som ikke er bevist}
}
