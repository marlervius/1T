\seksjon{4.1}{Gjennomsnittlig og momentan vekstfart}
\undertittel{Hvor fort endrer det seg? Fra to punkt til ett.}
\maal{Regne ut og tolke gjennomsnittlig vekstfart og anslå momentan vekstfart}
\kompetansemaal{10}

\hvakanjeg{
\kan{1}{regne ut gjennomsnittlig vekstfart fra formel, tabell og graf}
\kan{1}{tolke svaret med riktig enhet}
\kan{1}{forklare at gjennomsnittlig vekstfart er stigningstallet til en sekant}
\kanskille
\kan{2}{forklare hva momentan vekstfart er, og se forskjell på den og gjennomsnittlig vekstfart}
\kan{2}{anslå momentan vekstfart ved å regne på stadig mindre intervaller}
\kan{2}{lese av stigningstallet til en tangent og vurdere hvor nøyaktig avlesningen er}
}
