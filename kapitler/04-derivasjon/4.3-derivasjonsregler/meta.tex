\seksjon{4.3}{Derivasjonsregler for polynom}
\undertittel{Derivér raskt. Finn tangenten.}
\maal{Derivere polynomer, tolke den deriverte i kontekst og finne tangentlikninger}
\kompetansemaal{10}

\hvakanjeg{
\kan{1}{derivere polynomer med reglene, også etter å ha gangt ut eller forkortet}
\kan{1}{regne ut $f'(a)$ og tolke tallet som stigningstall og fart}
\kan{1}{forklare hvorfor vi ikke deriverer et produkt faktor for faktor}
\kanskille
\kan{2}{finne tangentlikningen i et punkt på formen $y = mx + c$}
\kan{2}{finne punkter der tangenten har et gitt stigningstall}
\kan{2}{forklare hvorfor tangentlikningen bruker tallet $f'(a)$}
}
