\seksjon{4.4}{Drøfting av funksjoner}
\undertittel{Les grafen av den deriverte. Finn størst og minst.}
\maal{Drøfte polynomfunksjoner med fortegnslinje for den deriverte, og finne største og minste verdi i praktiske problemer}
\kompetansemaal{11,10}

\hvakanjeg{
\kan{1}{lage fortegnslinje for $f'$ og lese av hvor $f$ er voksende og avtagende}
\kan{1}{finne topp-, bunn- og terrassepunkter og oppgi dem som koordinater}
\kan{1}{forklare hvorfor $f'(x) = 0$ ikke alltid gir et topp- eller bunnpunkt}
\kan{1}{drøfte et polynom, med nullpunkter fra polynomdivisjon, og skissere grafen}
\kanskille
\kan{2}{finne største og minste verdi på et lukket intervall}
\kan{2}{løse optimeringsoppgaver i fire steg}
\kan{2}{svare med en hel setning og enhet}
\kan{2}{forklare hvorfor endepunktene må sjekkes}
}
